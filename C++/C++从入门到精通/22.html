C++ 文件操作

文件读写、文本文件、二进制文件

文件操作概述

在C++编程中，文件操作是处理数据持久化存储的重要方式。通过文件操作，程序可以将数据保存到磁盘文件中，或者从文件中读取数据。C++标准库提供了丰富的文件操作功能，主要分为文本文件操作和二进制文件操作两大类。

文件流类

C++使用文件流类来处理文件输入输出操作，这些类定义在<fstream>头文件中：

类名 描述 主要用途
ifstream 输入文件流类 用于从文件读取数据
ofstream 输出文件流类 用于向文件写入数据
fstream 文件流类 既可读取也可写入文件
提示： 使用文件流类前，务必包含 <fstream> 头文件。

文本文件操作

文本文件以可读的字符形式存储数据，适合存储字符串、数字等人类可读的信息。

写入文本文件

#include <iostream>
#include <fstream>
using namespace std;

int main() {
    // 创建输出文件流对象
    ofstream outFile("example.txt");
    
    // 检查文件是否成功打开
    if (!outFile) {
        cout << "无法打开文件!" << endl;
        return 1;
    }
    
    // 向文件写入数据
    outFile << "Hello, World!" << endl;
    outFile << "这是C++文件操作示例" << endl;
    outFile << "数字: " << 42 << endl;
    outFile << "浮点数: " << 3.14159 << endl;
    
    // 关闭文件
    outFile.close();
    
    cout << "数据已写入文件!" << endl;
    return 0;
}

读取文本文件

#include <iostream>
#include <fstream>
#include <string>
using namespace std;

int main() {
    // 创建输入文件流对象
    ifstream inFile("example.txt");
    
    // 检查文件是否成功打开
    if (!inFile) {
        cout << "无法打开文件!" << endl;
        return 1;
    }
    
    string line;
    
    // 逐行读取文件内容
    while (getline(inFile, line)) {
        cout << line << endl;
    }
    
    // 关闭文件
    inFile.close();
    
    return 0;
}

二进制文件操作

二进制文件以原始字节形式存储数据，适合存储结构体、数组等复杂数据类型，效率更高。

写入二进制文件

#include <iostream>
#include <fstream>
using namespace std;

struct Student {
    char name[50];
    int age;
    double score;
};

int main() {
    Student students[3] = {
        {"张三", 20, 85.5},
        {"李四", 19, 92.0},
        {"王五", 21, 78.5}
    };
    
    // 以二进制模式打开文件用于写入
    ofstream outFile("students.dat", ios::binary);
    
    if (!outFile) {
        cout << "无法打开文件!" << endl;
        return 1;
    }
    
    // 写入数据
    for (int i = 0; i < 3; i++) {
        outFile.write((char*)&students[i], sizeof(Student));
    }
    
    outFile.close();
    cout << "二进制数据已写入文件!" << endl;
    
    return 0;
}

读取二进制文件

#include <iostream>
#include <fstream>
using namespace std;

struct Student {
    char name[50];
    int age;
    double score;
};

int main() {
    Student student;
    
    // 以二进制模式打开文件用于读取
    ifstream inFile("students.dat", ios::binary);
    
    if (!inFile) {
        cout << "无法打开文件!" << endl;
        return 1;
    }
    
    // 读取并显示数据
    while (inFile.read((char*)&student, sizeof(Student))) {
        cout << "姓名: " << student.name << endl;
        cout << "年龄: " << student.age << endl;
        cout << "成绩: " << student.score << endl;
        cout << "-------------------" << endl;
    }
    
    inFile.close();
    
    return 0;
}

文本文件 vs 二进制文件

文本文件

  • 以字符形式存储数据
  • 人类可读
  • 占用空间相对较大
  • 处理速度较慢
  • 适合存储字符串、配置信息
  • 使用 << 和 >> 运算符

二进制文件

  • 以字节形式存储数据
  • 人类不可读
  • 占用空间较小
  • 处理速度较快
  • 适合存储结构体、数组等
  • 使用 read() 和 write() 方法

文件打开模式

在打开文件时，可以指定不同的模式来控制文件的行为：

模式标志 描述
ios::in 以读取模式打开文件
ios::out 以写入模式打开文件
ios::app 以追加模式打开文件，在文件末尾添加数据
ios::trunc 如果文件已存在，清空文件内容
ios::binary 以二进制模式打开文件 
// 示例：组合使用文件打开模式
ofstream file;

// 以写入和追加模式打开文本文件
file.open("data.txt", ios::out | ios::app);

// 以读写二进制模式打开文件
fstream binFile("data.bin", ios::in | ios::out | ios::binary);

文件操作最佳实践

1
总是检查文件是否成功打开

在操作文件之前，务必检查文件流对象的状态，确保文件已成功打开。

2
及时关闭文件

完成文件操作后，及时调用close()方法关闭文件，释放系统资源。

3
选择合适的文件类型

根据数据类型和用途选择文本文件或二进制文件。

4
处理异常情况

考虑文件不存在、权限不足等异常情况，并提供相应的错误处理。

注意： 二进制文件在不同平台之间可能存在兼容性问题，特别是在处理结构体时要注意内存对齐和字节序问题。

综合示例：学生成绩管理系统

下面是一个完整的学生成绩管理系统示例，演示了文本文件和二进制文件的操作：

#include <iostream>
#include <fstream>
#include <vector>
#include <string>
using namespace std;

struct Student {
    string name;
    int age;
    double score;
};

// 向文本文件添加学生信息
void addStudentToTextFile(const Student& student, const string& filename) {
    ofstream file(filename, ios::app);
    if (file) {
        file << student.name << "," << student.age << "," << student.score << endl;
        file.close();
        cout << "学生信息已添加到文本文件" << endl;
    } else {
        cout << "无法打开文本文件" << endl;
    }
}

// 从文本文件读取学生信息
void readStudentsFromTextFile(const string& filename) {
    ifstream file(filename);
    if (file) {
        string line;
        cout << "从文本文件读取的学生信息:" << endl;
        while (getline(file, line)) {
            cout << line << endl;
        }
        file.close();
    } else {
        cout << "无法打开文本文件" << endl;
    }
}

// 向二进制文件添加学生信息
void addStudentToBinaryFile(const Student& student, const string& filename) {
    ofstream file(filename, ios::binary | ios::app);
    if (file) {
        // 写入姓名长度和姓名
        size_t nameLength = student.name.length();
        file.write((char*)&nameLength, sizeof(nameLength));
        file.write(student.name.c_str(), nameLength);
        
        // 写入年龄和成绩
        file.write((char*)&student.age, sizeof(student.age));
        file.write((char*)&student.score, sizeof(student.score));
        
        file.close();
        cout << "学生信息已添加到二进制文件" << endl;
    } else {
        cout << "无法打开二进制文件" << endl;
    }
}

// 从二进制文件读取学生信息
void readStudentsFromBinaryFile(const string& filename) {
    ifstream file(filename, ios::binary);
    if (file) {
        cout << "从二进制文件读取的学生信息:" << endl;
        while (file) {
            Student student;
            size_t nameLength;
            
            // 读取姓名长度
            file.read((char*)&nameLength, sizeof(nameLength));
            if (file.eof()) break;
            
            // 读取姓名
            char* nameBuffer = new char[nameLength + 1];
            file.read(nameBuffer, nameLength);
            nameBuffer[nameLength] = '\0';
            student.name = nameBuffer;
            delete[] nameBuffer;
            
            // 读取年龄和成绩
            file.read((char*)&student.age, sizeof(student.age));
            file.read((char*)&student.score, sizeof(student.score));
            
            cout << "姓名: " << student.name 
                 << ", 年龄: " << student.age 
                 << ", 成绩: " << student.score << endl;
        }
        file.close();
    } else {
        cout << "无法打开二进制文件" << endl;
    }
}

int main() {
    vector<Student> students = {
        {"小明", 18, 88.5},
        {"小红", 19, 92.0},
        {"小刚", 20, 76.5}
    };
    
    // 操作文本文件
    for (const auto& student : students) {
        addStudentToTextFile(student, "students.txt");
    }
    readStudentsFromTextFile("students.txt");
    
    cout << endl;
    
    // 操作二进制文件
    for (const auto& student : students) {
        addStudentToBinaryFile(student, "students.bin");
    }
    readStudentsFromBinaryFile("students.bin");
    
    return 0;
}
