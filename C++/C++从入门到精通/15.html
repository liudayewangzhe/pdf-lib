C++ 继承详解

第15章：单继承、多继承、访问权限、继承中的构造析构

1. 继承的基本概念

继承是面向对象编程的重要特性之一，它允许我们根据一个类来定义另一个类，使得代码可以重用，并且可以建立类之间的层次关系。

继承的优点：

  • 代码重用：可以复用基类的代码
  • 扩展性：可以在不修改基类的情况下扩展功能
  • 层次结构：可以建立清晰的类层次关系

2. 单继承

单继承是指一个派生类只从一个基类继承。这是最简单的继承形式。

2.1 单继承语法

class 基类 {
    // 基类成员
};

class 派生类 : 访问修饰符 基类 {
    // 派生类成员
};

2.2 单继承示例

#include <iostream>
using namespace std;

// 基类
class Animal {
protected:
    string name;
    int age;
    
public:
    Animal(string n, int a) : name(n), age(a) {}
    
    void eat() {
        cout << name << "正在吃东西" << endl;
    }
    
    void sleep() {
        cout << name << "正在睡觉" << endl;
    }
};

// 派生类
class Dog : public Animal {
private:
    string breed;
    
public:
    Dog(string n, int a, string b) : Animal(n, a), breed(b) {}
    
    void bark() {
        cout << name << "(" << breed << ")在汪汪叫" << endl;
    }
    
    void displayInfo() {
        cout << "名字: " << name << ", 年龄: " << age 
             << ", 品种: " << breed << endl;
    }
};

int main() {
    Dog myDog("旺财", 3, "金毛");
    myDog.eat();        // 继承自基类
    myDog.sleep();      // 继承自基类
    myDog.bark();       // 派生类自己的方法
    myDog.displayInfo();
    
    return 0;
}

输出结果：

旺财正在吃东西

旺财正在睡觉

旺财(金毛)在汪汪叫

名字: 旺财, 年龄: 3, 品种: 金毛

3. 访问权限

在继承中，访问权限控制着基类成员在派生类中的可见性。C++提供了三种继承方式：public、protected和private。

3.1 访问权限表

基类中的访问权限 public继承 protected继承 private继承
public public protected private
protected protected protected private
private 不可访问 不可访问 不可访问

3.2 访问权限示例

#include <iostream>
using namespace std;

class Base {
private:
    int privateVar;
protected:
    int protectedVar;
public:
    int publicVar;
    
    Base() : privateVar(1), protectedVar(2), publicVar(3) {}
    
    void showPrivate() {
        cout << "privateVar = " << privateVar << endl;
    }
};

class PublicDerived : public Base {
public:
    void accessBaseMembers() {
        // privateVar = 10;  // 错误：不能访问基类的private成员
        protectedVar = 20;   // 正确：可以访问基类的protected成员
        publicVar = 30;      // 正确：可以访问基类的public成员
        showPrivate();       // 正确：可以通过基类的public方法访问private成员
    }
};

class PrivateDerived : private Base {
public:
    void accessBaseMembers() {
        // privateVar = 10;  // 错误：不能访问基类的private成员
        protectedVar = 20;   // 正确：可以访问基类的protected成员（变为private）
        publicVar = 30;      // 正确：可以访问基类的public成员（变为private）
        showPrivate();       // 正确：可以通过基类的public方法访问private成员
    }
};

int main() {
    PublicDerived pubObj;
    pubObj.publicVar = 100;  // 正确：public继承后public成员仍然是public
    
    PrivateDerived privObj;
    // privObj.publicVar = 100;  // 错误：private继承后public成员变为private
    
    return 0;
}
💡 重要提示

在实际开发中，最常用的是public继承，因为它保持了"是一个"的关系。protected和private继承使用较少，它们表示"以...实现"的关系。

4. 多继承

多继承是指一个派生类可以从多个基类继承。这提供了更大的灵活性，但也可能带来复杂性。

4.1 多继承语法

class 基类1 {
    // 基类1成员
};

class 基类2 {
    // 基类2成员
};

class 派生类 : 访问修饰符 基类1, 访问修饰符 基类2 {
    // 派生类成员
};

4.2 多继承示例

#include <iostream>
#include <string>
using namespace std;

// 第一个基类
class Printer {
public:
    void print(string text) {
        cout << "打印: " << text << endl;
    }
};

// 第二个基类
class Scanner {
public:
    void scan() {
        cout << "正在扫描文档..." << endl;
    }
};

// 第三个基类
class Fax {
public:
    void sendFax(string message) {
        cout << "发送传真: " << message << endl;
    }
};

// 多继承的派生类
class MultiFunctionPrinter : public Printer, public Scanner, public Fax {
private:
    string model;
    
public:
    MultiFunctionPrinter(string m) : model(m) {}
    
    void showModel() {
        cout << "多功能打印机型号: " << model << endl;
    }
};

int main() {
    MultiFunctionPrinter mfp("Canon MX920");
    
    mfp.showModel();
    mfp.print("测试文档");      // 来自Printer基类
    mfp.scan();                // 来自Scanner基类
    mfp.sendFax("重要文件");   // 来自Fax基类
    
    return 0;
}

输出结果：

多功能打印机型号: Canon MX920

打印: 测试文档

正在扫描文档...

发送传真: 重要文件

⚠️ 注意：菱形继承问题

多继承可能带来菱形继承问题（钻石问题），即一个类通过多条路径继承同一个基类。这会导致二义性和数据冗余。可以使用虚继承来解决这个问题。

5. 继承中的构造和析构

在继承关系中，构造和析构函数的调用顺序有特定的规则。

5.1 构造函数调用顺序

  1. 基类的构造函数
  2. 成员对象的构造函数（按声明顺序）
  3. 派生类自己的构造函数

5.2 析构函数调用顺序

  1. 派生类自己的析构函数
  2. 成员对象的析构函数（按声明顺序的逆序）
  3. 基类的析构函数

5.3 构造析构示例

#include <iostream>
using namespace std;

class Base {
private:
    int baseData;
    
public:
    Base(int data) : baseData(data) {
        cout << "Base构造函数被调用, baseData = " << baseData << endl;
    }
    
    ~Base() {
        cout << "Base析构函数被调用, baseData = " << baseData << endl;
    }
};

class Member {
private:
    int memberData;
    
public:
    Member(int data) : memberData(data) {
        cout << "Member构造函数被调用, memberData = " << memberData << endl;
    }
    
    ~Member() {
        cout << "Member析构函数被调用, memberData = " << memberData << endl;
    }
};

class Derived : public Base {
private:
    Member m1;
    Member m2;
    int derivedData;
    
public:
    Derived(int b, int m1d, int m2d, int d) 
        : Base(b), m1(m1d), m2(m2d), derivedData(d) {
        cout << "Derived构造函数被调用, derivedData = " << derivedData << endl;
    }
    
    ~Derived() {
        cout << "Derived析构函数被调用, derivedData = " << derivedData << endl;
    }
};

int main() {
    cout << "创建Derived对象..." << endl;
    Derived obj(10, 20, 30, 40);
    
    cout << "\nmain函数结束，对象即将销毁..." << endl;
    return 0;
}

输出结果：

创建Derived对象...

Base构造函数被调用, baseData = 10

Member构造函数被调用, memberData = 20

Member构造函数被调用, memberData = 30

Derived构造函数被调用, derivedData = 40


main函数结束，对象即将销毁...

Derived析构函数被调用, derivedData = 40

Member析构函数被调用, memberData = 30

Member析构函数被调用, memberData = 20

Base析构函数被调用, baseData = 10

构造析构要点总结：

  • 构造函数调用顺序：基类 → 成员对象 → 派生类
  • 析构函数调用顺序：派生类 → 成员对象 → 基类（完全相反）
  • 如果基类没有默认构造函数，必须在派生类构造函数的初始化列表中显式调用基类构造函数
  • 析构函数应该是虚函数，特别是在有多态性的情况下

6. 综合示例

下面是一个综合示例，展示了继承的各种特性在实际中的应用。

#include <iostream>
#include <string>
#include <vector>
using namespace std;

// 基类：人员
class Person {
protected:
    string name;
    int age;
    
public:
    Person(string n, int a) : name(n), age(a) {
        cout << "创建Person: " << name << endl;
    }
    
    virtual ~Person() {
        cout << "销毁Person: " << name << endl;
    }
    
    virtual void display() const {
        cout << "姓名: " << name << ", 年龄: " << age;
    }
    
    string getName() const { return name; }
    int getAge() const { return age; }
};

// 派生类：学生
class Student : public Person {
private:
    string studentId;
    double gpa;
    
public:
    Student(string n, int a, string id, double g) 
        : Person(n, a), studentId(id), gpa(g) {
        cout << "创建Student: " << name << endl;
    }
    
    ~Student() {
        cout << "销毁Student: " << name << endl;
    }
    
    void display() const override {
        Person::display();
        cout << ", 学号: " << studentId << ", GPA: " << gpa << endl;
    }
    
    void study() {
        cout << name << "正在学习..." << endl;
    }
};

// 派生类：教师
class Teacher : public Person {
private:
    string employeeId;
    string department;
    
public:
    Teacher(string n, int a, string id, string dept) 
        : Person(n, a), employeeId(id), department(dept) {
        cout << "创建Teacher: " << name << endl;
    }
    
    ~Teacher() {
        cout << "销毁Teacher: " << name << endl;
    }
    
    void display() const override {
        Person::display();
        cout << ", 工号: " << employeeId << ", 部门: " << department << endl;
    }
    
    void teach() {
        cout << name << "正在授课..." << endl;
    }
};

int main() {
    vector<Person*> people;
    
    // 创建学生和教师对象
    people.push_back(new Student("张三", 20, "S001", 3.8));
    people.push_back(new Teacher("李老师", 35, "T001", "计算机系"));
    people.push_back(new Student("李四", 22, "S002", 3.5));
    
    cout << "\n=== 人员信息 ===" << endl;
    for (auto person : people) {
        person->display();
    }
    
    cout << "\n=== 特定操作 ===" << endl;
    // 向下转型以调用特定方法
    Student* stu = dynamic_cast<Student*>(people[0]);
    if (stu) {
        stu->study();
    }
    
    Teacher* tea = dynamic_cast<Teacher*>(people[1]);
    if (tea) {
        tea->teach();
    }
    
    cout << "\n=== 清理内存 ===" << endl;
    for (auto person : people) {
        delete person;
    }
    
    return 0;
}
