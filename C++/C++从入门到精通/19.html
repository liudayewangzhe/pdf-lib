C++ STL容器详解

vector、list、map、set容器使用指南

STL容器概述

STL（Standard Template Library，标准模板库）是C++标准库的重要组成部分，提供了一系列通用的模板类和函数，实现了常用的数据结构和算法。STL容器是其中最重要的组成部分，用于存储和管理数据集合。

注意：本课程重点讲解四种最常用的STL容器：vector、list、map和set。

vector容器

vector简介

vector 是一个动态数组，能够自动管理内存，支持随机访问，在尾部插入和删除元素效率高，但在中间或头部插入删除效率较低。

vector常用操作

操作 说明 时间复杂度
push_back() 在末尾添加元素 O(1) 平摊
pop_back() 删除末尾元素 O(1)
insert() 在指定位置插入元素 O(n)
erase() 删除指定位置元素 O(n)
operator[] 随机访问元素 O(1)
size() 返回元素个数 O(1)

vector示例代码

#include <iostream>
#include <vector>
using namespace std;

int main() {
    // 创建vector
    vector<int> nums = {1, 2, 3, 4, 5};
    
    // 添加元素
    nums.push_back(6);
    nums.push_back(7);
    
    // 遍历vector
    cout << "vector元素: ";
    for(int i = 0; i < nums.size(); i++) {
        cout << nums[i] << " ";
    }
    cout << endl;
    
    // 使用迭代器遍历
    cout << "使用迭代器: ";
    for(auto it = nums.begin(); it != nums.end(); it++) {
        cout << *it << " ";
    }
    cout << endl;
    
    // 删除元素
    nums.pop_back(); // 删除最后一个元素
    nums.erase(nums.begin() + 2); // 删除第三个元素
    
    // 范围for循环遍历
    cout << "删除后: ";
    for(int num : nums) {
        cout << num << " ";
    }
    cout << endl;
    
    return 0;
}

list容器

list简介

list 是一个双向链表，在任何位置插入和删除元素都非常高效，但不支持随机访问，只能通过迭代器顺序访问。

list常用操作

操作 说明 时间复杂度
push_back() 在末尾添加元素 O(1)
push_front() 在开头添加元素 O(1)
pop_back() 删除末尾元素 O(1)
pop_front() 删除开头元素 O(1)
insert() 在指定位置插入元素 O(1)
erase() 删除指定位置元素 O(1)
sort() 排序 O(n log n)

list示例代码

#include <iostream>
#include <list>
using namespace std;

int main() {
    // 创建list
    list<int> mylist = {5, 2, 8, 1, 9};
    
    // 添加元素
    mylist.push_back(10);
    mylist.push_front(0);
    
    // 遍历list
    cout << "list元素: ";
    for(auto it = mylist.begin(); it != mylist.end(); it++) {
        cout << *it << " ";
    }
    cout << endl;
    
    // 排序
    mylist.sort();
    
    cout << "排序后: ";
    for(int num : mylist) {
        cout << num << " ";
    }
    cout << endl;
    
    // 删除特定元素
    mylist.remove(5); // 删除所有值为5的元素
    
    // 在指定位置插入
    auto it = mylist.begin();
    advance(it, 2); // 移动到第三个位置
    mylist.insert(it, 100);
    
    cout << "最终结果: ";
    for(int num : mylist) {
        cout << num << " ";
    }
    cout << endl;
    
    return 0;
}

map容器

map简介

map 是一个关联容器，存储键值对，按键自动排序，每个键只能在map中出现一次。基于红黑树实现，查找、插入和删除操作的时间复杂度为O(log n)。

map常用操作

操作 说明 时间复杂度
insert() 插入键值对 O(log n)
erase() 删除键值对 O(log n)
find() 查找键 O(log n)
operator[] 访问或插入元素 O(log n)
count() 统计键出现次数 O(log n)
size() 返回元素个数 O(1)

map示例代码

#include <iostream>
#include <map>
#include <string>
using namespace std;

int main() {
    // 创建map
    map<string, int> scores;
    
    // 插入元素
    scores["Alice"] = 95;
    scores["Bob"] = 87;
    scores["Charlie"] = 92;
    scores.insert({"David", 78});
    
    // 遍历map
    cout << "学生成绩:" << endl;
    for(auto& pair : scores) {
        cout << pair.first << ": " << pair.second << endl;
    }
    
    // 查找元素
    string name = "Bob";
    auto it = scores.find(name);
    if(it != scores.end()) {
        cout << name << "的成绩是: " << it->second << endl;
    } else {
        cout << "未找到" << name << "的成绩" << endl;
    }
    
    // 使用[]访问元素（如果不存在会创建）
    cout << "Eva的成绩: " << scores["Eva"] << endl;
    
    // 统计元素个数
    cout << "map中有 " << scores.size() << " 个元素" << endl;
    
    // 删除元素
    scores.erase("Charlie");
    
    cout << "删除Charlie后:" << endl;
    for(auto& pair : scores) {
        cout << pair.first << ": " << pair.second << endl;
    }
    
    return 0;
}

set容器

set简介

set 是一个关联容器，存储唯一元素，自动排序。基于红黑树实现，查找、插入和删除操作的时间复杂度为O(log n)。

set常用操作

操作 说明 时间复杂度
insert() 插入元素 O(log n)
erase() 删除元素 O(log n)
find() 查找元素 O(log n)
count() 统计元素出现次数 O(log n)
size() 返回元素个数 O(1)

set示例代码

#include <iostream>
#include <set>
using namespace std;

int main() {
    // 创建set
    set<int> numbers = {5, 2, 8, 1, 9, 2, 5}; // 重复元素会被自动去重
    
    // 遍历set
    cout << "set中的元素: ";
    for(int num : numbers) {
        cout << num << " ";
    }
    cout << endl;
    
    // 插入元素
    numbers.insert(3);
    numbers.insert(7);
    
    // 查找元素
    int target = 8;
    auto it = numbers.find(target);
    if(it != numbers.end()) {
        cout << "找到元素 " << target << endl;
    } else {
        cout << "未找到元素 " << target << endl;
    }
    
    // 统计元素
    cout << "set中有 " << numbers.size() << " 个元素" << endl;
    
    // 删除元素
    numbers.erase(5);
    
    cout << "删除5后: ";
    for(int num : numbers) {
        cout << num << " ";
    }
    cout << endl;
    
    // 使用迭代器删除
    it = numbers.find(9);
    if(it != numbers.end()) {
        numbers.erase(it);
    }
    
    cout << "删除9后: ";
    for(int num : numbers) {
        cout << num << " ";
    }
    cout << endl;
    
    return 0;
}

容器比较与选择

vector

  • 动态数组
  • 随机访问高效
  • 尾部插入删除高效
  • 中间插入删除低效
  • 内存连续

list

  • 双向链表
  • 任意位置插入删除高效
  • 不支持随机访问
  • 内存不连续
  • 额外指针开销

map

  • 键值对存储
  • 按键自动排序
  • 基于红黑树
  • 查找高效
  • 键唯一

set

  • 唯一元素存储
  • 自动排序
  • 基于红黑树
  • 查找高效
  • 元素唯一

选择指南：

  • 需要随机访问 → vector
  • 频繁在任意位置插入删除 → list
  • 需要键值对映射 → map
  • 需要存储唯一元素并快速查找 → set

综合示例

以下是一个综合使用多种STL容器的示例，展示如何在实际问题中灵活运用这些容器：

#include <iostream>
#include <vector>
#include <list>
#include <map>
#include <set>
#include <string>
#include <algorithm>
using namespace std;

// 学生信息结构体
struct Student {
    string name;
    int id;
    double score;
    
    Student(string n, int i, double s) : name(n), id(i), score(s) {}
    
    // 重载<运算符，用于set排序
    bool operator<(const Student& other) const {
        return id < other.id;
    }
};

int main() {
    // 使用vector存储学生对象
    vector<Student> students;
    students.push_back(Student("Alice", 101, 95.5));
    students.push_back(Student("Bob", 102, 87.0));
    students.push_back(Student("Charlie", 103, 92.5));
    
    // 使用map按科目存储成绩
    map<string, vector<double>> subjectScores;
    subjectScores["Math"] = {95, 87, 92, 78, 85};
    subjectScores["English"] = {88, 92, 79, 95, 82};
    subjectScores["Science"] = {90, 85, 88, 92, 87};
    
    // 使用set存储唯一的学生ID
    set<int> studentIDs;
    for(const auto& student : students) {
        studentIDs.insert(student.id);
    }
    
    // 使用list存储待处理的学生
    list<Student> pendingStudents;
    pendingStudents.push_back(Student("David", 104, 79.0));
    pendingStudents.push_back(Student("Eva", 105, 88.5));
    
    // 输出学生信息
    cout << "学生信息:" << endl;
    for(const auto& student : students) {
        cout << "姓名: " << student.name << ", ID: " << student.id 
             << ", 成绩: " << student.score << endl;
    }
    
    // 输出科目平均分
    cout << "\n科目平均分:" << endl;
    for(const auto& subject : subjectScores) {
        double sum = 0;
        for(double score : subject.second) {
            sum += score;
        }
        double average = sum / subject.second.size();
        cout << subject.first << ": " << average << endl;
    }
    
    // 输出唯一的学生ID
    cout << "\n唯一的学生ID:" << endl;
    for(int id : studentIDs) {
        cout << id << " ";
    }
    cout << endl;
    
    // 处理待处理学生
    cout << "\n待处理学生:" << endl;
    for(const auto& student : pendingStudents) {
        cout << "姓名: " << student.name << ", ID: " << student.id 
             << ", 成绩: " << student.score << endl;
    }
    
    return 0;
}
