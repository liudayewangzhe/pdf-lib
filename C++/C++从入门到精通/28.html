C++静态成员详解

静态数据成员、静态成员函数、静态常量

静态成员概述

在C++中，静态成员是类的成员，它们与类本身相关联，而不是与类的特定对象相关联。静态成员在类的所有对象之间共享，无论创建了多少个对象，静态成员都只有一份副本。

静态成员分为三种类型：

  • 静态数据成员 - 类的所有对象共享的数据
  • 静态成员函数 - 不依赖于特定对象的函数
  • 静态常量 - 在类中定义的常量值

静态数据成员

静态数据成员是类的所有对象共享的数据成员。无论创建了多少个类的对象，静态数据成员在内存中只有一份副本。

声明和定义

在类内部声明静态数据成员，在类外部进行定义和初始化：

class MyClass {
public:
    static int count;  // 声明静态数据成员
    // ...
};

int MyClass::count = 0;  // 定义并初始化静态数据成员

访问静态数据成员

可以通过类名或对象访问静态数据成员：

// 通过类名访问
MyClass::count = 10;

// 通过对象访问
MyClass obj1, obj2;
obj1.count = 20;
obj2.count = 30;  // 所有对象共享同一个count

cout << MyClass::count;  // 输出30

示例：使用静态数据成员统计对象数量

#include <iostream>
using namespace std;

class Student {
private:
    string name;
    int age;
public:
    static int count;  // 静态数据成员，用于统计学生数量
    
    Student(string n, int a) : name(n), age(a) {
        count++;  // 每创建一个学生对象，计数器加1
    }
    
    ~Student() {
        count--;  // 对象销毁时，计数器减1
    }
};

// 定义并初始化静态数据成员
int Student::count = 0;

int main() {
    cout << "初始学生数量: " << Student::count << endl;
    
    Student s1("张三", 18);
    Student s2("李四", 19);
    
    cout << "创建两个学生后的数量: " << Student::count << endl;
    
    {
        Student s3("王五", 20);
        cout << "在内部块中创建第三个学生后的数量: " << Student::count << endl;
    }
    
    cout << "内部块结束后学生数量: " << Student::count << endl;
    
    return 0;
}

静态成员函数

静态成员函数是不依赖于特定对象的成员函数。它们只能访问静态数据成员和其他静态成员函数，不能访问非静态成员。

声明和定义

class MyClass {
private:
    static int count;
    int normalData;
public:
    static int getCount() {  // 静态成员函数
        return count;
        // 错误：不能访问normalData，因为它是非静态成员
    }
};

调用静态成员函数

可以通过类名或对象调用静态成员函数：

// 通过类名调用
int c = MyClass::getCount();

// 通过对象调用
MyClass obj;
int c2 = obj.getCount();

示例：银行账户类

#include <iostream>
using namespace std;

class BankAccount {
private:
    string owner;
    double balance;
    static double interestRate;  // 静态数据成员，所有账户共享的利率
    
public:
    BankAccount(string o, double b) : owner(o), balance(b) {}
    
    // 静态成员函数，用于设置利率
    static void setInterestRate(double rate) {
        interestRate = rate;
    }
    
    // 静态成员函数，用于获取利率
    static double getInterestRate() {
        return interestRate;
    }
    
    // 非静态成员函数，计算利息
    double calculateInterest() {
        return balance * interestRate;
    }
    
    void display() {
        cout << "账户持有人: " << owner << endl;
        cout << "余额: " << balance << endl;
        cout << "利率: " << interestRate * 100 << "%" << endl;
        cout << "利息: " << calculateInterest() << endl;
    }
};

// 定义并初始化静态数据成员
double BankAccount::interestRate = 0.03;

int main() {
    BankAccount::setInterestRate(0.035);  // 通过类名调用静态成员函数
    
    BankAccount acc1("张三", 5000);
    BankAccount acc2("李四", 8000);
    
    acc1.display();
    cout << "-------------------" << endl;
    acc2.display();
    
    return 0;
}

静态常量

静态常量是在类中定义的常量值，它们与静态数据成员类似，但值在编译时确定且不能修改。

整数静态常量

对于整数类型的静态常量，可以在类内部直接初始化：

class MathConstants {
public:
    static const int MAX_VALUE = 100;  // 整数静态常量
    static const double PI;  // 非整数静态常量需要在类外定义
};

const double MathConstants::PI = 3.14159;

使用静态常量

class Array {
private:
    int data[Array::MAX_SIZE];
public:
    static const int MAX_SIZE = 100;  // 静态常量
    
    Array() {
        // 使用静态常量作为数组大小
    }
};

示例：使用静态常量

#include <iostream>
using namespace std;

class Circle {
private:
    double radius;
public:
    // 静态常量
    static const double PI;
    static const int DEFAULT_RADIUS = 5;
    
    Circle(double r = DEFAULT_RADIUS) : radius(r) {}
    
    double area() {
        return PI * radius * radius;
    }
    
    double circumference() {
        return 2 * PI * radius;
    }
    
    void display() {
        cout << "半径: " << radius << endl;
        cout << "面积: " << area() << endl;
        cout << "周长: " << circumference() << endl;
    }
};

// 定义静态常量PI
const double Circle::PI = 3.1415926535;

int main() {
    Circle c1;  // 使用默认半径
    Circle c2(10);  // 使用自定义半径
    
    cout << "圆1:" << endl;
    c1.display();
    
    cout << "\n圆2:" << endl;
    c2.display();
    
    cout << "\nπ的值: " << Circle::PI << endl;
    cout << "默认半径: " << Circle::DEFAULT_RADIUS << endl;
    
    return 0;
}

静态成员特性总结

特性 静态数据成员 静态成员函数 静态常量
存储 所有对象共享一份副本 不依赖于对象存在 所有对象共享一份副本
访问 通过类名或对象访问 通过类名或对象调用 通过类名或对象访问
初始化 在类外部定义和初始化 在类内部定义 整数类型可在类内初始化，其他需在类外定义
访问权限 可访问任何成员 只能访问静态成员 可访问任何成员
内存分配 程序开始时分配 程序开始时分配 程序开始时分配

关键要点

  • 静态成员属于类本身，而不是类的对象
  • 静态数据成员用于存储类的所有对象共享的数据
  • 静态成员函数不能访问非静态成员，因为它们没有this指针
  • 静态常量提供了一种在类中定义常量的方式
  • 静态数据成员必须在类外部定义和初始化（除了整数静态常量）
  • 静态成员可以通过类名直接访问，无需创建对象

注意：静态成员函数没有this指针，因此不能是虚函数，也不能使用const、volatile或ref限定符。

综合示例

下面是一个综合示例，展示了静态成员的各种用法：

#include <iostream>
#include <string>
using namespace std;

class Car {
private:
    string brand;
    string model;
    int year;
    
    // 静态数据成员
    static int totalCars;
    
public:
    // 静态常量
    static const int MAX_YEAR = 2023;
    static const int MIN_YEAR = 1886;  // 第一辆汽车发明的年份
    
    // 构造函数
    Car(string b, string m, int y) : brand(b), model(m), year(y) {
        if (year >= MIN_YEAR && year <= MAX_YEAR) {
            totalCars++;
        } else {
            cout << "错误：年份无效！" << endl;
        }
    }
    
    // 析构函数
    ~Car() {
        totalCars--;
    }
    
    // 静态成员函数
    static int getTotalCars() {
        return totalCars;
    }
    
    static void displayCarInfo() {
        cout << "汽车信息：" << endl;
        cout << "最早年份: " << MIN_YEAR << endl;
        cout << "最晚年份: " << MAX_YEAR << endl;
        cout << "总汽车数量: " << totalCars << endl;
    }
    
    // 非静态成员函数
    void display() const {
        cout << "品牌: " << brand << endl;
        cout << "型号: " << model << endl;
        cout << "年份: " << year << endl;
    }
};

// 定义静态数据成员
int Car::totalCars = 0;

int main() {
    // 使用静态常量和静态成员函数，无需创建对象
    cout << "汽车最早年份: " << Car::MIN_YEAR << endl;
    cout << "汽车最晚年份: " << Car::MAX_YEAR << endl;
    cout << "初始汽车数量: " << Car::getTotalCars() << endl;
    
    cout << "\n创建汽车对象..." << endl;
    Car car1("丰田", "卡罗拉", 2020);
    Car car2("本田", "思域", 2021);
    Car car3("福特", "福克斯", 2022);
    
    cout << "当前汽车数量: " << Car::getTotalCars() << endl;
    
    cout << "\n汽车详细信息：" << endl;
    car1.display();
    cout << "---" << endl;
    car2.display();
    cout << "---" << endl;
    car3.display();
    
    cout << "\n静态函数显示信息：" << endl;
    Car::displayCarInfo();
    
    return 0;
}
