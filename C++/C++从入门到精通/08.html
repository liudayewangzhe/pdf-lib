1. 函数定义与声明

函数是C++程序的基本构建块，用于封装可重用的代码。一个函数由函数头和函数体组成。

函数声明（函数原型）

函数声明告诉编译器函数的名称、返回类型和参数，但不包含函数体。

// 函数声明
返回类型 函数名(参数列表);

示例：

// 函数声明
int add(int a, int b);
double calculateArea(double radius);
void printMessage();

函数定义

函数定义包含函数的实际实现，即函数体。

// 函数定义
返回类型 函数名(参数列表) {
    // 函数体
    // 执行语句
    return 返回值; // 如果返回类型不是void
}

示例：

// 函数定义
int add(int a, int b) {
    int result = a + b;
    return result;
}

double calculateArea(double radius) {
    return 3.14159 * radius * radius;
}

void printMessage() {
    cout << "Hello, World!" << endl;
}
提示： 在C++中，函数必须先声明后使用。通常将函数声明放在头文件中，函数定义放在源文件中。

2. 参数传递

C++支持三种参数传递方式：值传递、引用传递和指针传递。

传递方式 语法 特点 适用场景
值传递 类型 参数名 创建参数副本，函数内修改不影响原变量 不需要修改原变量的情况
引用传递 类型 &参数名 直接操作原变量，函数内修改会影响原变量 需要修改原变量或传递大型对象
指针传递 类型 *参数名 通过地址操作原变量，函数内修改会影响原变量 需要修改原变量或处理动态内存

参数传递示例

#include <iostream>
using namespace std;

// 值传递
void swapByValue(int a, int b) {
    int temp = a;
    a = b;
    b = temp;
    cout << "函数内: a = " << a << ", b = " << b << endl;
}

// 引用传递
void swapByReference(int &a, int &b) {
    int temp = a;
    a = b;
    b = temp;
    cout << "函数内: a = " << a << ", b = " << b << endl;
}

// 指针传递
void swapByPointer(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
    cout << "函数内: *a = " << *a << ", *b = " << *b << endl;
}

int main() {
    int x = 5, y = 10;
    
    cout << "原始值: x = " << x << ", y = " << y << endl;
    
    swapByValue(x, y);
    cout << "值传递后: x = " << x << ", y = " << y << endl;
    
    swapByReference(x, y);
    cout << "引用传递后: x = " << x << ", y = " << y << endl;
    
    swapByPointer(&x, &y);
    cout << "指针传递后: x = " << x << ", y = " << y << endl;
    
    return 0;
}
注意： 值传递会创建参数的副本，对于大型对象可能会影响性能。在这种情况下，应使用引用传递或指针传递。

3. 返回值

函数可以通过return语句返回一个值。返回值的类型必须与函数声明中的返回类型一致。

返回值类型

  • 基本数据类型：int, double, char, bool等
  • 自定义类型：结构体、类对象等
  • 指针类型：返回内存地址
  • 引用类型：返回变量的引用
  • void类型：不返回任何值

返回值示例

#include <iostream>
#include <string>
using namespace std;

// 返回基本数据类型
int getMax(int a, int b) {
    return (a > b) ? a : b;
}

// 返回字符串
string getGrade(int score) {
    if (score >= 90) return "优秀";
    else if (score >= 80) return "良好";
    else if (score >= 70) return "中等";
    else if (score >= 60) return "及格";
    else return "不及格";
}

// 返回引用
int& getElement(int arr[], int index) {
    return arr[index];
}

// 返回指针
int* createArray(int size) {
    int* arr = new int[size];
    for (int i = 0; i < size; i++) {
        arr[i] = i * i;
    }
    return arr;
}

// 无返回值
void printArray(int arr[], int size) {
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}

int main() {
    // 使用返回基本数据类型的函数
    int maxVal = getMax(10, 20);
    cout << "最大值: " << maxVal << endl;
    
    // 使用返回字符串的函数
    string grade = getGrade(85);
    cout << "成绩等级: " << grade << endl;
    
    // 使用返回引用的函数
    int numbers[] = {1, 2, 3, 4, 5};
    getElement(numbers, 2) = 100; // 修改数组元素
    cout << "修改后的数组: ";
    printArray(numbers, 5);
    
    // 使用返回指针的函数
    int* dynamicArray = createArray(5);
    cout << "动态数组: ";
    printArray(dynamicArray, 5);
    delete[] dynamicArray; // 释放内存
    
    return 0;
}
最佳实践： 当函数需要返回多个值时，可以考虑使用引用参数或返回结构体/类对象。

4. 函数重载

函数重载允许在同一作用域内定义多个同名函数，只要它们的参数列表不同（参数类型、参数个数或参数顺序不同）。

函数重载规则

  1. 函数名必须相同
  2. 参数列表必须不同（类型、个数或顺序）
  3. 返回类型可以相同也可以不同，但仅返回类型不同不足以构成重载
  4. 重载函数应该在相似的操作上使用，以提高代码可读性

函数重载示例

#include <iostream>
#include <string>
using namespace std;

// 重载函数：打印整数
void print(int value) {
    cout << "整数: " << value << endl;
}

// 重载函数：打印浮点数
void print(double value) {
    cout << "浮点数: " << value << endl;
}

// 重载函数：打印字符串
void print(const string& value) {
    cout << "字符串: " << value << endl;
}

// 重载函数：打印两个整数
void print(int a, int b) {
    cout << "两个整数: " << a << " 和 " << b << endl;
}

// 重载函数：打印整数数组
void print(int arr[], int size) {
    cout << "整数数组: ";
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}

// 计算面积的重载函数
double calculateArea(double radius) { // 圆形面积
    return 3.14159 * radius * radius;
}

double calculateArea(double width, double height) { // 矩形面积
    return width * height;
}

double calculateArea(double base, double height, bool isTriangle) { // 三角形面积
    if (isTriangle) {
        return 0.5 * base * height;
    }
    return base * height; // 如果不是三角形，返回矩形面积
}

int main() {
    // 调用不同的print重载函数
    print(10);
    print(3.14);
    print("Hello, C++!");
    print(5, 8);
    
    int numbers[] = {1, 2, 3, 4, 5};
    print(numbers, 5);
    
    // 调用不同的calculateArea重载函数
    cout << "圆形面积: " << calculateArea(5.0) << endl;
    cout << "矩形面积: " << calculateArea(4.0, 6.0) << endl;
    cout << "三角形面积: " << calculateArea(4.0, 6.0, true) << endl;
    
    return 0;
}
注意： 函数重载与返回类型无关。仅返回类型不同的函数不能构成重载，会导致编译错误。

5. 综合示例

下面是一个综合示例，演示了函数定义、参数传递、返回值和函数重载的实际应用。

#include <iostream>
#include <string>
#include <cmath>
using namespace std;

// 函数声明
double calculate(double a, double b, char operation);
double calculate(double radius);
void displayResult(double result, const string& message = "结果");

// 使用引用参数修改多个值
void calculateCircle(double radius, double& area, double& circumference);

// 函数重载：计算阶乘
long long factorial(int n);
double factorial(double n); // 使用Gamma函数近似计算实数的阶乘

int main() {
    // 基本计算器功能
    double num1, num2;
    char op;
    
    cout << "请输入两个数字和运算符(+, -, *, /): ";
    cin >> num1 >> num2 >> op;
    
    double result = calculate(num1, num2, op);
    displayResult(result, "计算结果");
    
    // 计算圆面积
    double radius;
    cout << "请输入圆的半径: ";
    cin >> radius;
    
    double circleResult = calculate(radius);
    displayResult(circleResult, "圆面积");
    
    // 使用引用参数返回多个值
    double area, circumference;
    calculateCircle(radius, area, circumference);
    
    cout << "圆的面积: " << area << endl;
    cout << "圆的周长: " << circumference << endl;
    
    // 计算阶乘
    int n;
    cout << "请输入一个整数计算阶乘: ";
    cin >> n;
    
    long long fact = factorial(n);
    cout << n << "的阶乘是: " << fact << endl;
    
    // 计算实数的阶乘（近似值）
    double x;
    cout << "请输入一个实数计算阶乘(近似值): ";
    cin >> x;
    
    double approxFact = factorial(x);
    cout << x << "的阶乘近似值是: " << approxFact << endl;
    
    return 0;
}

// 函数定义：基本计算器
double calculate(double a, double b, char operation) {
    switch(operation) {
        case '+':
            return a + b;
        case '-':
            return a - b;
        case '*':
            return a * b;
        case '/':
            if(b != 0) {
                return a / b;
            } else {
                cout << "错误：除数不能为零!" << endl;
                return 0;
            }
        default:
            cout << "错误：不支持的运算符!" << endl;
            return 0;
    }
}

// 函数重载：计算圆面积
double calculate(double radius) {
    return 3.14159 * radius * radius;
}

// 函数定义：显示结果
void displayResult(double result, const string& message) {
    cout << message << ": " << result << endl;
}

// 函数定义：计算圆的面积和周长
void calculateCircle(double radius, double& area, double& circumference) {
    area = 3.14159 * radius * radius;
    circumference = 2 * 3.14159 * radius;
}

// 函数定义：计算整数阶乘
long long factorial(int n) {
    if (n < 0) {
        cout << "错误：阶乘不支持负数!" << endl;
        return -1;
    }
    if (n == 0 || n == 1) {
        return 1;
    }
    
    long long result = 1;
    for (int i = 2; i <= n; i++) {
        result *= i;
    }
    return result;
}

// 函数重载：计算实数阶乘（使用斯特林公式近似）
double factorial(double n) {
    if (n < 0) {
        cout << "错误：阶乘不支持负数!" << endl;
        return -1;
    }
    
    // 斯特林公式近似
    return sqrt(2 * 3.14159 * n) * pow(n / 2.71828, n);
}
学习建议： 多练习编写函数，理解不同参数传递方式的区别，熟练掌握函数重载的使用场景。
